#include <iostream>

using namespace std;

int is_pair(char t, char inp){
    if (t=='[' && inp==']'){
        return 3;
    }
    if (t=='(' && inp==')'){
        return 2;
    }
    return 0;
}

int is_vaild(string t){
    stack<char> s;
    for(auto&e:t){
        if (!s.empty() && is_pair(s.top(),e)){
            s.pop();
        } else {
            s.push(e);
        }
    }
    if (s.empty()){
        return 1;
    }
    return 0;
}
